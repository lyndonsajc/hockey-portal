rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    match /announcements/{id} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    match /attendance/{sessionId}/records/{studentId} {
      allow read: if true;
      allow create, update: if
        request.resource.data.keys().hasOnly([
          'studentId','name','className','group','gender','status','reason',
          'sessionId','sessionLabel','updatedAt','timestamp'
        ])
        && request.resource.data.status in [
          'Present','Absent','MC/Parent Letter','Unmarked'
        ];
      allow delete: if request.auth != null;
    }

    match /teacherAttendanceRecords/{recordId} {
      allow read: if request.auth != null;
      allow create, update: if
        request.resource.data.keys().hasOnly([
          'sessionId','sessionLabel','date','day','start','end','venue','remarks',
          'studentId','name','className','group','gender','status','reason',
          'updatedAt','timestamp'
        ])
        && request.resource.data.status in [
          'Present','Absent','MC/Parent Letter','Unmarked'
        ];
      allow delete: if request.auth != null;
    }

    match /attendanceSessions/{sessionId} {
      allow read: if true;
      allow create, update: if
        request.resource.data.keys().hasOnly([
          'sessionId','sessionLabel','date','day','start','end','venue','remarks',
          'updatedAt','timestamp'
        ]);
      allow delete: if request.auth != null;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
